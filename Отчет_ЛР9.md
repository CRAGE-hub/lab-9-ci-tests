## 4. Настройка CI

Workflow запускается на каждом `push` и `pull_request`, а также поддерживает ручной запуск `workflow_dispatch`. Для сборки используется `ubuntu-latest`. В CI устанавливаются Python 3.12 и пакет pytest из корневого `requirements.txt`, затем выполняется `python -m pytest -v`. Разрешение `contents: read` ограничивает права job необходимым чтением репозитория. Кэширование pip использует файл зависимостей.

Полный конфигурационный файл:

### `.github/workflows/run-tests.yml`

```yaml
name: Python tests

on:
  push:
  pull_request:
  workflow_dispatch:

permissions:
  contents: read

jobs:
  pytest:
    name: Run pytest
    runs-on: ubuntu-latest
    steps:
      - name: Check out repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: pip
          cache-dependency-path: requirements.txt

      - name: Install dependencies
        run: python -m pip install -r requirements.txt

      - name: Run tests
        run: python -m pytest -v
```

## 6. Публикация и завершающая проверка в GitHub

1. Создать на GitHub **публичный** репозиторий (например, `lab-9-ci-tests`). Не включать создание README, если будете загружать уже подготовленные файлы этой папки.
2. Загрузить содержимое папки `9` в корень нового репозитория, сохранив путь `.github/workflows/run-tests.yml`.
3. Выполнить commit/push в основную ветку. Событие `push` автоматически запустит workflow.
4. Открыть вкладку **Actions**, выбрать запуск **Python tests** и проверить, что job **Run pytest** завершился успешно.
5. При необходимости запустить workflow вручную через **Run workflow**. Если запуск красный — открыть упавший step, исправить причину, снова сделать push и убедиться, что последний запуск зеленый.

## 7. Исходный код разработанного приложения

### `calculator.py`

```python
class Calculator:
    """Простые арифметические операции и проверка простых чисел."""

    def add(self, left: int | float, right: int | float) -> int | float:
        return left + right

    def subtract(self, left: int | float, right: int | float) -> int | float:
        return left - right

    def divide(self, numerator: int | float, denominator: int | float) -> float:
        if denominator == 0:
            raise ZeroDivisionError("Деление на ноль невозможно.")
        return numerator / denominator

    def is_prime_number(self, number: int) -> bool:
        if isinstance(number, bool) or not isinstance(number, int):
            raise TypeError("Проверяемое значение должно быть целым числом.")
        if number < 2:
            return False

        divisor = 2
        while divisor * divisor <= number:
            if number % divisor == 0:
                return False
            divisor += 1
        return True
```

## 8. Исходный код автоматических тестов

### `tests/test_calculator.py`

```python
import pytest

from calculator import Calculator


@pytest.fixture
def calculator() -> Calculator:
    return Calculator()


@pytest.mark.parametrize(
    ("left", "right", "expected"),
    [
        (2, 3, 5),
        (0, 0, 0),
        (-2, 5, 3),
        (1.25, 2.5, 3.75),
    ],
)
def test_add(calculator: Calculator, left: int | float, right: int | float, expected: int | float) -> None:
    assert calculator.add(left, right) == expected


@pytest.mark.parametrize(
    ("left", "right", "expected"),
    [
        (8, 3, 5),
        (0, 7, -7),
        (-4, -6, 2),
        (1.5, 2.25, -0.75),
    ],
)
def test_subtract(calculator: Calculator, left: int | float, right: int | float, expected: int | float) -> None:
    assert calculator.subtract(left, right) == expected


@pytest.mark.parametrize(
    ("numerator", "denominator", "expected"),
    [
        (8, 2, 4.0),
        (7, 2, 3.5),
        (-9, 3, -3.0),
        (0, 5, 0.0),
    ],
)
def test_divide(calculator: Calculator, numerator: int | float, denominator: int | float, expected: float) -> None:
    assert calculator.divide(numerator, denominator) == expected


@pytest.mark.parametrize("denominator", [0, 0.0, -0.0])
def test_divide_by_zero_raises_error(calculator: Calculator, denominator: int | float) -> None:
    with pytest.raises(ZeroDivisionError, match="Деление на ноль невозможно"):
        calculator.divide(10, denominator)


@pytest.mark.parametrize(
    ("number", "expected"),
    [
        (2, True),
        (3, True),
        (4, False),
        (1, False),
        (0, False),
        (-7, False),
        (97, True),
        (100, False),
        (7919, True),
        (49, False),
    ],
)
def test_is_prime_number(calculator: Calculator, number: int, expected: bool) -> None:
    assert calculator.is_prime_number(number) is expected


@pytest.mark.parametrize("value", [2.5, "7", True, None])
def test_is_prime_number_rejects_non_integers(calculator: Calculator, value: object) -> None:
    with pytest.raises(TypeError, match="целым числом"):
        calculator.is_prime_number(value)
```
